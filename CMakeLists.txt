cmake_minimum_required(VERSION 3.16)
project(l1w_loco_manipulation_deploy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/lowlevel/lowlevel.h")
    message(FATAL_ERROR "Vendor lowlevel.h is required. Obtain it from the vendor and place it at include/lowlevel/lowlevel.h. See include/lowlevel/README.md.")
endif()
add_executable(lowlevel_bridge src/lowlevel_bridge.cpp)
target_include_directories(lowlevel_bridge PRIVATE "include")
target_compile_options(lowlevel_bridge PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(lowlevel_bridge PRIVATE pthread rt)
